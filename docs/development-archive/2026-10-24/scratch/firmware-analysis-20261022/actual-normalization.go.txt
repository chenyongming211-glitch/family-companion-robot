package main
import("unicode";"strings";"encoding/json";"os")
func normalizeWakePhrase(text string) string {
	return strings.Map(func(r rune) rune {
		if unicode.IsSpace(r) || unicode.IsPunct(r) {
			return -1
		}
		return unicode.ToLower(r)
	}, text)
}
