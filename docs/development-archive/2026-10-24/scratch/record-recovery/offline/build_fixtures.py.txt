import subprocess,wave,json,hashlib,array,math
from pathlib import Path
ROOT=Path(__file__).resolve().parent
SR=16000
D=ROOT/'fixtures';D.mkdir(exist_ok=True)
def load(p):
 with wave.open(str(p)) as f:
  assert (f.getframerate(),f.getnchannels(),f.getsampwidth())==(SR,1,2)
  return array.array('h',f.readframes(f.getnframes()))
def save(p,x):
 with wave.open(str(p),'wb') as f:f.setparams((1,2,SR,0,'NONE','not compressed'));f.writeframes(x.tobytes())
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
sets={'count':['一','二','三','四','五','六','七','八','九','十'], 'ordinary':['今天阳光很好','我们一起去公园','然后回家吃晚饭'], 'terms':['明天下午三点半','在上海虹桥站','检查OpenStack服务','版本二点三']}
cases=[];sources={}
def case(name,source,start=0,end=None,draft=None,note=''):
 s=sources[source];end=len(s['pcm']) if end is None else end
 p=D/(name+'.wav');save(p,s['pcm'][start:end]);truth=[]
 for u in s['units']:
  if u['start_frame']>=start and u['end_frame']<=end:
   truth.append(dict(u,start=(u['start_frame']-start)/SR,end=(u['end_frame']-start)/SR))
 c={'name':name,'audio':str(p.relative_to(ROOT)),'text':draft or s['text'],'audio_complete':start==0 and end==len(s['pcm']),'source':source,'source_sha256':s['sha256'],'source_start_frame':start,'source_end_frame':end,'sample_rate':SR,'audio_sha256':sha(p),'truth_units':truth,'full_source_units':s['units'],'note':note,'identity':{'generation':1,'response_id':name,'item_id':name,'content_index':0}}
 cases.append(c)
for name,units in sets.items():
 pcm=array.array('h',[0]*int(.4*SR));labels=[]
 for i,text in enumerate(units):
  p=D/f'{name}-unit-{i}.wav'
  subprocess.run(['say','-v','Tingting','-r','150','-o',str(p),'--data-format=LEI16@16000',text],check=True)
  raw=load(p);active=[j for j,x in enumerate(raw) if abs(x)>120]
  # Independent source support from amplitude; no evaluated model involved.
  lo=max(0,min(active)-320);hi=min(len(raw),max(active)+321);x=raw[lo:hi]
  st=len(pcm);pcm.extend(x);en=len(pcm)
  labels.append({'text':text,'index':i,'start_frame':st,'end_frame':en,'unit_file_sha256':sha(p),'raw_trim_start_frame':lo,'raw_trim_end_frame':hi})
  pcm.extend([0]*int(.32*SR))
 p=D/(name+'-source.wav');save(p,pcm)
 sources[name]={'pcm':pcm,'units':labels,'text':'，'.join(units)+'。','sha256':sha(p)}
 case(name+'-full',name)
case('early-silence-full-draft','count',end=int(.2*SR),note='cancel in independently inserted leading silence; correct complete units=0')
c=sources['count']['units'];case('early-first-unit-full-draft','count',end=c[0]['start_frame']+(c[0]['end_frame']-c[0]['start_frame'])//4,note='cancel at quarter of first source unit; complete units=0; phonetic partial word unknown')
case('count-gap-after-three-full-draft','count',end=c[2]['end_frame']+int(.16*SR),note='cancel at known inserted gap after three')
case('count-mid-five-full-draft','count',end=(c[4]['start_frame']+c[4]['end_frame'])//2,note='cut inside source unit five, truth complete through four')
o=sources['ordinary']['units'];case('ordinary-mid-second-full-draft','ordinary',end=(o[1]['start_frame']+o[1]['end_frame'])//2,note='sentence middle at constructed phrase midpoint; no phoneme boundary claim')
case('ordinary-gap-after-two-full-draft','ordinary',end=o[1]['end_frame']+int(.16*SR))
case('resume-mid-again-full-remainder','count',start=c[3]['start_frame'],end=(c[6]['start_frame']+c[6]['end_frame'])//2,draft='，'.join(sets['count'][3:])+'。',note='explicit synthetic resume from four; interrupted again inside seven; complete units four/five/six; old source coordinates rebased')
t=sources['terms']['units'];case('terms-mid-third-full-draft','terms',end=(t[2]['start_frame']+t[2]['end_frame'])//2)
# All-zero with full future draft, kept separate from early 200ms silence.
p=D/'silence.wav';save(p,array.array('h',[0]*(3*SR)))
cases.append({'name':'silence-full-draft','audio':str(p.relative_to(ROOT)),'text':sources['count']['text'],'audio_complete':True,'audio_sha256':sha(p),'truth_units':[],'sample_rate':SR,'note':'three seconds constructed silence; zero expected speech units','identity':{'generation':1,'response_id':'silence','item_id':'silence','content_index':0}})
# Existing native continuous output: source provenance, no independently labelled word truth.
native=Path('.superpowers/resume-state-alignment/manifest.json');old=json.loads(native.read_text())
for oldname in ['native-raw-arabic','native-raw-arabic-truncated','native-explicit-chinese-numerals']:
 c=next(x for x in old['cases'] if x['name']==oldname).copy();p=(native.parent/c['audio']).resolve();c['audio']=str(p);c['audio_sha256']=sha(p);c['note']='native provider continuous output; no independent unit truth; do not score timing';cases.append(c)
manifest={'generator':'macOS say Tingting rate150, unit clips amplitude-trim abs>120 +/-320 frames at16k then .32s inserted silence; .4s leading silence; no playback/mic','truth_contract':'Independently constructed clip supports and inserted silence, not aligner labels. Unit identity is synthesis script label, phonetic realization not manually audited. (ROOT/'manifest.json').write_text(json.dumps(manifest,ensure_ascii=False,indent=2)+'\n')
print([(c['name'],len(c.get('truth_units',[]))) for c in cases])
