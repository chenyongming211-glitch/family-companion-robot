import json,hashlib,re,unicodedata,statistics
from pathlib import Path
R=Path(__file__).resolve().parent
M=json.loads((R/'manifest.json').read_text());cases={c['name']:c for c in M['cases']}
cn=['零','一','二','三','四','五','六','七','八','九','十']
def num(n):
 n=int(n)
 return cn[n] if n<11 else ('十'+cn[n%10] if n<20 else cn[n//10]+'十'+(cn[n%10] if n%10 else '')) if n<=50 else str(n)
variant=str.maketrans('陽們園後飯點檢務橋紅歐', '阳们园后饭点检务桥红欧')
def norm(t,rich=True):
 if rich:
  t=t.translate(variant)
  t=re.sub(r'(?<!\d)(\d+)\.(\d+)(?!\d)',lambda m:num(m[1])+'点'+''.join(cn[int(x)] for x in m[2]),t)
  t=re.sub(r'\d+',lambda m:num(m[0]),t)
 return ''.join(x.lower() for x in t if not x.isspace() and not unicodedata.category(x).startswith('P'))
def edits(a,b):
 p=list(range(len(b)+1))
 for i,x in enumerate(a):
  q=[i+1]
  for j,y in enumerate(b):q.append(min(q[-1]+1,p[j+1]+1,p[j]+(x!=y)))
  p=q
 return p[-1]
def groups(c,r):
 source=c.get('full_source_units',[])
 if c['name']=='silence-full-draft':source=[{'index':i,'text':x} for i,x in enumerate(M['sources']['count']['units'][j]['text'] for j in range(10))]
 if c.get('source_start_frame',0):source=[u for u in source if u['start_frame']>=c['source_start_frame']]
 spans=r.get('spans',[]);out=[];cursor=0
 for u in source:
  text=norm(u['text']);found=None
  for a in range(cursor,len(spans)):
   for b in range(a,len(spans)):
    if norm(''.join(x['text'] for x in spans[a:b+1]))==text:
     found=(a,b);break
   if found:break
  if found is None:continue
  a,b=found;p=spans[a:b+1]
  out.append({'index':u['index'],'text':u['text'],'start':p[0]['start'],'end':p[-1]['end'],'positive_constituents':all(x['end']>x['start'] for x in p)})
  cursor=b+1
 return out

  if 'truth_units' in c:
   gs=groups(c,r);truth=c['truth_units'];truth_ids=[x['index'] for x in truth]
   # Identity-based completion set; matched fully recognized unit ending within clip.
   picked=[g for g in gs if g['end']<=d and g['end']>g['start']]
