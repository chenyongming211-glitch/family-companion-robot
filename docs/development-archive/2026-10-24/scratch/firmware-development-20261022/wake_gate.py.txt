bool AfeAudioEngine::GetWakeWordOpus(std::vector<uint8_t>& opus) {
    if (wake_detector_==WakeDetector::kMultiNet) return custom_wake_word_->GetWakeWordOpus(opus);
    if (wake_detector_!=WakeDetector::kWakeNet) return false;
    std::unique_lock<std::mutex> lock(wake_word_mutex_);
    wake_word_cv_.wait(lock,[this] {return !wake_word_opus_.empty();});
    auto packet=std::move(wake_word_opus_.front());
    wake_word_opus_.pop_front();
    if (packet.batch!=wake_batch_.load() ||
        (codec_->SoftwareReference() && packet.generation!=codec_->WakeAudioGeneration())) {
        opus.clear(); return false;
    }
    opus=std::move(packet.opus);
    return !opus.empty();
}
