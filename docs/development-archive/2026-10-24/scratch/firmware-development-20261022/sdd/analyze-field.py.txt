def correlation(a,b,lags,minimum=1000):
 if len(a)<minimum or len(b)<minimum or not np.any(a) or not np.any(b):return None
 n=len(a)+len(b)-1;nf=1<<(n-1).bit_length();v=np.fft.irfft(np.fft.rfft(a,nf)*np.conj(np.fft.rfft(b,nf)),nf);cs=np.r_[0,np.cumsum(a*a)];ys=np.r_[0,np.cumsum(b*b)];best=None
