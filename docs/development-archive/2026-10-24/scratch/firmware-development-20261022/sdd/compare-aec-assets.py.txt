def digest(b):return hashlib.sha256(b).hexdigest()
def parse(data):
 n,check,length=struct.unpack_from('<III',data)
 assert 0<n<=1024 and n*44<=length and 12+length<=len(data)
 assert sum(data[12:12+length])&65535==check
 entries={};payloads={}
 for i in range(n):
  raw,size,offset,width,height=struct.unpack_from('<32sIIHH',data,12+i*44)
  name=raw.split(b'\0')[0].decode();start=12+n*44+offset
  assert name not in entries and start>=12+n*44 and start+2+size<=12+length
  assert data[start:start+2]==b'ZZ'
  content=data[start+2:start+2+size]
  entries[name]={'bytes':size,'sha256':digest(content),'width':width,'height':height}
  payloads[name]=content
 return {'files':n,'used_bytes':12+length,'checksum_valid':True,'entries':entries},payloads

def models(data):

   assert key not in entries and offset+size<=len(data)
   entries[key]={'bytes':size,'sha256':digest(data[offset:offset+size])}
