const fixedClarificationTest = "我还不能确认刚才停在哪儿。请告诉我从哪里继续，或从头再来。"

func clarificationHarness(t *testing.T) (*playbackHarness, *toolPeer, map[string]any) {
	t.Helper()
	h := newPlaybackHarness(t)
	s := h.s
	s.responseID = ""
	s.generation = 1
	s.currentInput = requestOrigin{generation: 1, itemID: "continue", known: true, text: "继续"}
	s.inputCommitted = true
	s.paused = &pausedTask{id: "task-1", original: requestOrigin{known: true, text: "从一数到五十"}, playback: playbackRecord{id: 8, settled: true}}
	s.snapshotRevision = 3
	t.Cleanup(s.cancelRouting)
	p := attachSnapshotModel(t, s)
	requests, results := controlledClassifier(s)
	if err := s.requestResponse(); err != nil {
		t.Fatal(err)
	}
	takeClassification(t, requests)
	results <- taskRouteResult{action: "resume"}
	if err := s.finishRouting(routeResult(t, s)); err != nil {
		t.Fatal(err)
	}
	p.read(t)
	ackSnapshotWire(t, s, p)
	create := p.want(t, "response.create")
	if err := s.onModel(context.Background(), qwen.Event{Type: "response.created", Response: qwen.Response{ID: "clarify"}}); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(s.encoder.Close)
	return h, p, create
}
func clarificationEvent(t *testing.T, s *session, ev qwen.Event) error {
	t.Helper()
	if ev.ResponseID == "" && ev.Type != "response.done" {
		ev.ResponseID = "clarify"
	}
	return s.onModel(context.Background(), ev)
}
func clarificationPCM(t *testing.T, s *session, pcm []byte) error {
	t.Helper()
	return clarificationEvent(t, s, qwen.Event{Type: "response.audio.delta", Delta: base64.StdEncoding.EncodeToString(pcm)})
}

func TestUnknownResumeValidClarificationReleasesOnlyAfterCompletion(t *testing.T) {
	h, _, _ := clarificationHarness(t)
	s := h.s
	original := s.paused.original
	if err := clarificationPCM(t, s, make([]byte, 2*audio.FrameBytes)); err != nil {
		t.Fatal(err)
	}
	for _, typ := range []string{"response.audio_transcript.delta", "response.audio_transcript.done"} {
		ev := qwen.Event{Type: typ, Delta: fixedClarificationTest, Transcript: fixedClarificationTest}
		if err := clarificationEvent(t, s, ev); err != nil {
			t.Fatal(err)
		}
		if got := h.play(t, time.Now()); len(got) != 0 {
			t.Fatalf("premature release at %s: %v", typ, got)
		}
	}
	if err := clarificationEvent(t, s, qwen.Event{Type: "response.audio.done"}); err != nil {
		t.Fatal(err)
	}
	if err := clarificationEvent(t, s, qwen.Event{Type: "response.done", Response: qwen.Response{ID: "clarify", Status: "completed"}}); err != nil {
		t.Fatal(err)
	}
	now := time.Now()
	if got := h.play(t, now); !reflect.DeepEqual(got, []string{"start", "sentence_start", "audio"}) {
		t.Fatal(got)
	}
	if got := h.play(t, now.Add(59*time.Millisecond)); len(got) != 0 {
		t.Fatalf("cadence escaped: %v", got)
	}
	if got := h.play(t, now.Add(60*time.Millisecond)); !reflect.DeepEqual(got, []string{"audio"}) {
		t.Fatal(got)
	}
	h.play(t, now.Add(120*time.Millisecond))
	if s.paused == nil || s.paused.original != original || s.paused.playback.id != 1 || s.pausedData()["text_position"] != "unknown" {
		t.Fatalf("clarification lost task: %v", s.pausedData())
	}
}
func TestUnknownResumeRejectsContaminatedClarificationBeforeDevice(t *testing.T) {
	for _, kind := range []string{"wrong-transcript", "extra-words", "text", "tool", "status", "zero", "odd", "overlong", "missing-transcript", "missing-audio-done", "delta-mismatch", "duplicate-done", "late-audio"} {
		t.Run(kind, func(t *testing.T) {
			h, _, _ := clarificationHarness(t)
			s := h.s
			original := s.paused.original
			pcm := make([]byte, audio.FrameBytes)
			if kind == "zero" {
				pcm = nil
			}
			if kind == "odd" {
				pcm = append(pcm, 0)
			}
			if kind == "overlong" {
				pcm = make([]byte, 720002)
			}
			err := clarificationPCM(t, s, pcm)
			transcript := fixedClarificationTest
			if kind == "wrong-transcript" {
				transcript = "一二三四五"
			}
			if kind == "extra-words" {
				transcript += "好的"
			}
			if err == nil && kind == "text" {
				err = clarificationEvent(t, s, qwen.Event{Type: "response.text.delta", Delta: fixedClarificationTest})
			}
			if err == nil && kind == "tool" {
				err = clarificationEvent(t, s, qwen.Event{Type: "response.function_call_arguments.done", CallID: "x", Name: "conversation_get_paused_task", Arguments: `{}`})
			}
			if err == nil && kind == "delta-mismatch" {
				err = clarificationEvent(t, s, qwen.Event{Type: "response.audio_transcript.delta", Delta: "一二三"})
			}
			if err == nil && kind != "missing-transcript" {
				err = clarificationEvent(t, s, qwen.Event{Type: "response.audio_transcript.done", Transcript: transcript})
			}
			if err == nil && kind == "duplicate-done" {
				err = clarificationEvent(t, s, qwen.Event{Type: "response.audio_transcript.done", Transcript: transcript})
			}
			if err == nil && kind != "missing-audio-done" {
				err = clarificationEvent(t, s, qwen.Event{Type: "response.audio.done"})
			}
			if err == nil && kind == "late-audio" {
				err = clarificationPCM(t, s, pcm)
			}
			if err == nil {
				status := "completed"
				if kind == "status" {
					status = "failed"
				}
				err = clarificationEvent(t, s, qwen.Event{Type: "response.done", Response: qwen.Response{ID: "clarify", Status: status}})
			}
			if err == nil {
				t.Fatal("invalid native clarification accepted")
			}
			if got := h.play(t, time.Now()); len(got) != 0 {
				t.Fatalf("rejected audio escaped: %v", got)
			}
			if s.pending.Len() != 0 || s.paused == nil || s.paused.original != original || len(s.toolCalls) > 0 {
				t.Fatal("rejected output retained PCM, mutated goal, or executed tool")
			}
		})
	}
}

func TestUnknownResumeDeadlineAndBoundedPCM(t *testing.T) {
	for _, phase := range []string{"before-created", "while-held", "at-done", "limit"} {
		t.Run(phase, func(t *testing.T) {
			h, _, _ := clarificationHarness(t)
			s := h.s
			if phase == "limit" {
				if err := clarificationPCM(t, s, make([]byte, maxClarificationPCM)); err != nil {
					t.Fatal(err)
				}
				if err := clarificationEvent(t, s, qwen.Event{Type: "response.audio_transcript.done", Transcript: fixedClarificationTest}); err != nil {
					t.Fatal(err)
				}
				clarificationEvent(t, s, qwen.Event{Type: "response.audio.done"})
				if err := clarificationEvent(t, s, qwen.Event{Type: "response.done", Response: qwen.Response{ID: "clarify", Status: "completed"}}); err != nil {
					t.Fatal(err)
				}
				return
			}
			if err := clarificationPCM(t, s, make([]byte, audio.FrameBytes)); err != nil {
				t.Fatal(err)
			}
			deadline := s.clarification.deadline
			if deadline.IsZero() || time.Until(deadline) > clarificationTimeout {
				t.Fatal("missing response.create deadline")
			}
			if phase == "before-created" {
				s.responseID = ""
				s.clarification.responseID = ""
			}
			var err error
			if phase == "at-done" {
				s.clarification.deadline = time.Now().Add(-time.Second)
				err = clarificationEvent(t, s, qwen.Event{Type: "response.done", Response: qwen.Response{ID: "clarify", Status: "completed"}})
			} else {
				h.now = deadline
				err = s.play()
			}
			if err == nil || s.pending.Len() != 0 || s.clarification.active {
				t.Fatal("deadline retained or released held PCM")
			}
			if got := h.play(t, deadline.Add(time.Second)); len(got) != 0 {
				t.Fatal(got)
			}
		})
	}
}

				err = clarificationEvent(t, s, qwen.Event{Type: "response.audio.done"})

				}

func TestUnknownResumeChoiceFactSurvivesSideAndClearsOnExplicitDecision(t *testing.T) {
	for _, action := range []string{"restart", "resume_from", "discard", "replace"} {
		t.Run(action, func(t *testing.T) {
			h, p, _ := clarificationHarness(t)
			s := h.s
			if s.pausedData()["awaiting_resume_choice"] != true {
				t.Fatal("question did not record gateway choice fact")
			}
			original := s.paused.original
			taskID := s.paused.id
			for i, route := range []string{"side", action} {
				s.commitOrigin(route)
				s.transcribeOrigin(route, "三十七")
				requests, results := controlledClassifier(s)
				if err := s.requestResponse(); err != nil {
					t.Fatal(err)
				}
				request := takeClassification(t, requests)
				if !request.awaiting {
					t.Fatal("classifier lost choice fact")
				}
				results <- taskRouteResult{action: route}
				if err := s.finishRouting(routeResult(t, s)); err != nil {
					t.Fatal(err)
				}
				cfg := p.read(t).body["session"].(map[string]any)
				if route == "side" {
					if s.paused.awaitingResumeChoice != true || s.responseTaskID != "" || s.paused.original != original {
						t.Fatal("side changed task or choice")
					}
					if !strings.Contains(cfg["instructions"].(string), `"awaiting_resume_choice":true`) {
						t.Fatal("snapshot omitted choice")
					}
				} else {
					if s.paused != nil && s.paused.awaitingResumeChoice {
						t.Fatal("explicit choice retained pending question")
					}
					if (route == "restart" || route == "resume_from") && (s.responseTaskID != taskID || s.paused.original != original) {
						t.Fatal("explicit start changed original goal")
					}
				}
				ackSnapshotWire(t, s, p)
				create := p.want(t, "response.create")
				if route == "restart" || route == "resume_from" {
					if create["response"] == nil {
						t.Fatal("explicit choice lacks response policy")
					}
				}
				if err := s.onModel(context.Background(), qwen.Event{Type: "response.created", Response: qwen.Response{ID: route}}); err != nil {
					t.Fatal(err)
				}
				t.Cleanup(s.encoder.Close)
				s.responseDone = true
				h.play(t, time.Now().Add(time.Duration(i)*time.Second))
			}
		})
	}
}

	if s.pending.Len() != 0 || s.clarification.active || s.paused.original.text != "从一数到五十" {
		t.Fatal("interruption retained old audio or lost goal")
	}
}

	s, p := routingHarness(t)
	requests, results := controlledClassifier(s)
	s.requestResponse()

	results <- taskRouteResult{action: "resume"}

	s.paused.awaitingResumeChoice = false
	s.snapshotRevision++
	ackSnapshotWire(t, s, p)

		t.Fatal(update)
	}

}
