#include "playback_progress.h"
#include <cassert>
#include <cstdio>
int main() {
  using namespace family_robot;
  PlaybackProgress p;
  const PlaybackIdentity id{1,1};
  int16_t pcm[240]{};
  int16_t dma[6][240]{};
  p.Connect(1); assert(p.Start(id));
  for (int i=0;i<3;++i) assert(p.Admit(id,i*1440,1440));
  for (int i=0;i<12;++i) assert(p.Submit(id,i*240,pcm,240)==PlaybackProgress::SubmitResult::Accepted);
  p.Release(id); p.Release(id); // Third admitted frame still awaits decode/output.
  p.Seal(id,false); // The normal WebSocket stop only seals.
  for (int i=0;i<3;++i) p.Complete(dma[i],240,(i+1)*10000);
  assert(!p.Snapshot(id).finished);
  p.InterruptAll(); // Local AbortSpeaking -> ResetDecoder.
  for (int i=3;i<9;++i) p.Complete(dma[i%6],240,(i+1)*10000);
  const auto result=p.Snapshot(id);
  assert(result.finished && result.valid && result.interrupted && result.played_samples==720);
  std::printf("sealed-before-interrupt: valid=%d finished=%d interrupted=%d played=%llu received=4320\n",result.valid,result.finished,result.interrupted,(unsigned long long)result.played_samples);
}
