// Isolated fixtures model the existing 4 protocol / 14 primary intents, including
// the one consumed v1 generated response. They never edit the real ledger.
func ledgerV2(t *testing.T) string {
	t.Helper()
	p := filepath.Join(t.TempDir(), "attempts.jsonl")
	f, e := os.OpenFile(p, os.O_CREATE|os.O_EXCL|os.O_WRONLY, 0600)
	if e != nil {
		t.Fatal(e)
	}
	defer f.Close()
	for stage, n := range map[string]int{"protocol": 4, "primary": 13} {
		for i := 0; i < n; i++ {
			json.NewEncoder(f).Encode(map[string]any{"event": "intent", "stage": stage, "attempt": fmt.Sprintf("old_%s_%d", stage, i)})
		}
	}
	json.NewEncoder(f).Encode(map[string]any{"event": "intent", "stage": "primary", "attempt": "generated-audio-repair-v1/completed-generate", "state": "outcome_unknown"})
	return p
}

type fakeV2 struct {
	mu                        sync.Mutex
	creates, cancels, deletes int
	violation                 string
}

func serverV2(t *testing.T, mode string) (string, *fakeV2) {
	t.Helper()
	f := &fakeV2{}
	s := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		c, e := (&websocket.Upgrader{}).Upgrade(w, r, nil)
		if e != nil {
			return
		}
		defer c.Close()
		c.SetReadDeadline(time.Now().Add(5 * time.Second))
		seq := 0
		write := func(v map[string]any) {
			seq++
			v["event_id"] = fmt.Sprintf("srv_%04d", seq)
			if mode == "duplicate_event_id" && v["type"] == "response.audio.delta" {
				v["event_id"] = "srv_0001"
			}
			c.WriteJSON(v)
		}
		write(map[string]any{"type": "session.created", "session": map[string]any{"id": "sess_v2"}})
		items := map[string]map[string]any{}
		deleted := map[string]bool{}
		last := ""
		active := 0
		for {
			var v map[string]any
			if c.ReadJSON(&v) != nil {
				return
			}
			kind := str(v["type"])
			switch kind {
			case "session.update":
				ss := obj(v["session"])
				ss["id"] = "sess_v2"
				ss["model"] = modelName
				write(map[string]any{"type": "session.updated", "session": ss})
			case "conversation.item.create":
				item := obj(v["item"])
				id := str(item["id"])
				prev := str(v["previous_item_id"])
				if prev == "" {
					prev = last
				}
				items[id] = item
				last = id
				if mode == "wrong_position" && item["role"] == "assistant" {
					prev = "foreign"
				}
				write(map[string]any{"type": "conversation.item.created", "previous_item_id": prev, "item": item})
			case "response.create":
				f.mu.Lock()
				f.creates++
				n := f.creates
				f.mu.Unlock()
				if n > 3 {
					f.mu.Lock()
					f.violation = "more_than_three_primary"
					f.mu.Unlock()
					return
				}
				active = n
				rid, id := fmt.Sprintf("r%d", n), fmt.Sprintf("a%d", n)
				text := playedPrefix
				if n == 2 {
					text += "紫贝壳藏在银色灯塔。"
					switch mode {
					case "nonce_substring":
						text = playedPrefix + "海风。"
					case "nonce_repeated_prefix":
						text = playedPrefix + playedPrefix
					case "nonce_contains_prefix":
						text = playedPrefix + playedPrefix + "银色灯塔。"
					}
				}
				if n == 3 && mode == "independent_nonce_leak" {
					text += "紫贝壳藏在银色灯塔。"
				}
				item := map[string]any{"id": id, "type": "message", "role": "assistant", "content": []any{map[string]any{"type": "audio", "transcript": text}}}
				items[id] = item
				if n == 2 || n == 3 {
					if !deleted[fmt.Sprintf("a%d", n-1)] {
						f.mu.Lock()
						f.violation = "generation_before_exact_delete"
						f.mu.Unlock()
					}
				}
				write(map[string]any{"type": "response.created", "response": map[string]any{"id": rid, "status": "in_progress", "output": []any{}}})
				write(map[string]any{"type": "response.output_item.added", "response_id": rid, "item": item})
				write(map[string]any{"type": "conversation.item.created", "previous_item_id": last, "item": item})
				last = id
				write(map[string]any{"type": "response.audio.delta", "response_id": rid, "item_id": id, "content_index": 0, "delta": base64.StdEncoding.EncodeToString(make([]byte, 4800))})
				write(map[string]any{"type": "response.audio_transcript.done", "response_id": rid, "item_id": id, "content_index": 0, "transcript": text})
				if n == 1 {
					continue
				}
				if mode == "multipart" && n == 3 {
					item = map[string]any{"id": id, "type": "message", "role": "assistant", "content": []any{map[string]any{"type": "audio", "transcript": "紫贝壳藏在银色灯塔。"}, map[string]any{"type": "audio", "transcript": playedPrefix}}}
				}
				output := []any{item}
				if mode == "extra_terminal_item" {
					output = append(output, map[string]any{"id": "foreign", "type": "message", "role": "assistant", "content": []any{}})
				}
				write(map[string]any{"type": "response.done", "response": map[string]any{"id": rid, "status": "completed", "output": output}})
			case "response.cancel":
				f.mu.Lock()
				f.cancels++
				f.mu.Unlock()
				valid := active == 1
				for k := range v {
					if k != "type" && k != "event_id" {
						valid = false
					}
				}
				if !valid {
					f.mu.Lock()
					f.violation = "wrong_cancel"
					f.mu.Unlock()
					return
				}
				if mode == "cancel_timeout" {
					continue
				}
				write(map[string]any{"type": "response.done", "response": map[string]any{"id": "r1", "status": "cancelled", "output": []any{items["a1"]}}})
			case "conversation.item.delete":
				id := str(v["item_id"])
				f.mu.Lock()
				f.deletes++
				f.mu.Unlock()
				deleted[id] = true
				if mode == "wrong_delete_ack" {
					id = "foreign"
				}
				write(map[string]any{"type": "conversation.item.deleted", "item_id": id})
			case "conversation.item.retrieve":
				id := str(v["item_id"])
				if deleted[id] {
					if mode == "retrieve_timeout" {
						continue
					}
					if mode == "returned_deleted_item" {
						write(map[string]any{"type": "conversation.item.retrieved", "item": items[id]})
						continue
					}
					if mode == "foreign_control" {
						write(map[string]any{"type": "conversation.item.deleted", "item_id": "foreign"})
						continue
					}
					err := map[string]any{"type": "invalid_request_error", "code": "invalid_value", "event_id": v["event_id"], "message": "synthetic generic invalid value"}
					switch mode {
					case "no_correlation":
						delete(err, "event_id")
					case "wrong_correlation":
						err["event_id"] = "foreign_request"
					case "server_error":
						err["type"] = "server_error"
					case "other_code":
						err["code"] = "rate_limit_exceeded"
					case "exact_absence":
						err["message"] = "Item " + id + " not found"
					}
					write(map[string]any{"type": "error", "error": err})
				} else {
					write(map[string]any{"type": "conversation.item.retrieved", "item": items[id]})
					if strings.HasSuffix(id, "_prefix") {
						write(map[string]any{"type": "conversation.item.input_audio_transcription.completed", "item_id": "ga_cancel_user", "transcript": "known late synthetic ASR"})
					}
				}
			default:
				return
			}
		}
	}))
	t.Cleanup(s.Close)
	return "ws" + strings.TrimPrefix(s.URL, "http"), f
}

func TestV2ExactlyThreeResponsesReuseCompletedFollowup(t *testing.T) {
	u, f := serverV2(t, "normal")
	out := filepath.Join(t.TempDir(), "protocol-v2")
	ledger := ledgerV2(t)
	r, e := runProbe(context.Background(), probeOptions{URL: u, Ledger: ledger, Output: out, StepTimeout: time.Second})
	if e != nil {
		t.Fatal(e)
	}
	if r.State != "protocol_acknowledged" || len(r.Scenarios) != 2 {
		t.Fatalf("result %+v", r)
	}
	a, b := r.Scenarios[0], r.Scenarios[1]
	if a.Name != "cancel" || a.Generated.Terminal != "cancelled" || !a.Generated.CancelSent || b.Name != "completed" || b.UserItem != "ga_cancel_followup" || b.Generated.ID != a.Followup.ID || b.Generated.ID != "r2" || b.Followup.ID != "r3" {
		t.Fatalf("completed response not reused: first %+v second %+v", a, b)
	}
	if b.HiddenSyntheticTail != "紫贝壳藏在银色灯塔。" || b.Behavior != "prefix_only_observed_internal_history_unknown" {
		t.Fatalf("behavior %+v", b)
	}
	f.mu.Lock()
	defer f.mu.Unlock()
	if f.creates != 3 || f.cancels != 1 || f.deletes != 2 || f.violation != "" {
		t.Fatalf("fake %+v", f)
	}
	bb, _ := os.ReadFile(ledger)
	if strings.Count(string(bb), attemptPrefix) != 4 {
		t.Fatalf("want protocol+3 independent primary intents: %s", bb)
	}
	for _, s := range r.Scenarios {
		if !s.Repaired {
			t.Fatal("missing repair gate")
		}
		for _, ev := range s.Evidence {
			if strings.Contains(ev.Kind, "absence") {
				t.Fatalf("generic invalid_value falsely granted absence: %+v", ev)
			}
			if ev.Kind == "exact_deleted_ACK" && (ev.LookupStatus != "absent_lookup_UNKNOWN" || ev.LookupFacts == nil) {
				t.Fatalf("optional lookup quality missing: %+v", ev)
			}
		}
	}
	bb, _ = os.ReadFile(filepath.Join(out, "events.jsonl"))
	if !strings.Contains(string(bb), "absent_lookup_UNKNOWN") || strings.Contains(string(bb), "synthetic generic invalid value") {
		t.Fatalf("missing bounded unknown evidence or raw error retained: %s", bb)
	}
}

func TestV2OptionalLookupNeverGrantsErasure(t *testing.T) {
	for _, mode := range []string{"normal", "no_correlation", "exact_absence"} {
		t.Run(mode, func(t *testing.T) {
			u, _ := serverV2(t, mode)
			r, e := runProbe(context.Background(), probeOptions{URL: u, Ledger: ledgerV2(t), Output: filepath.Join(t.TempDir(), "protocol-v2"), StepTimeout: time.Second})
			if e != nil || r.State != "protocol_acknowledged" {
				t.Fatalf("optional evidence stopped valid ACK repair: %+v %v", r, e)
			}
			for _, sc := range r.Scenarios {
				ev := sc.Evidence[0]
				wantCorr := "matched"
				if mode == "no_correlation" {
					wantCorr = "absent"
				}
				if ev.Kind != "exact_deleted_ACK" || ev.LookupStatus != "absent_lookup_UNKNOWN" || ev.LookupFacts == nil || ev.LookupFacts.Code != "invalid_value" || ev.LookupFacts.Type != "invalid_request_error" || ev.LookupFacts.Correlation != wantCorr {
					t.Fatalf("optional error falsely graded: %+v", ev)
				}
				if mode == "exact_absence" && (!ev.LookupFacts.ExactItemInMessage || !ev.LookupFacts.AbsencePhraseInMessage) {
					t.Fatalf("bounded facts lost: %+v", ev.LookupFacts)
				}
			}
		})
	}
}

func TestV2HardUnknownsStop(t *testing.T) {
	for _, mode := range []string{"wrong_correlation", "server_error", "other_code", "returned_deleted_item", "retrieve_timeout", "foreign_control", "wrong_delete_ack", "wrong_position", "cancel_timeout", "duplicate_event_id", "extra_terminal_item", "multipart"} {
		t.Run(mode, func(t *testing.T) {
			u, f := serverV2(t, mode)
			r, e := runProbe(context.Background(), probeOptions{URL: u, Ledger: ledgerV2(t), Output: filepath.Join(t.TempDir(), "protocol-v2"), StepTimeout: 100 * time.Millisecond})
			if e == nil || r.State != "unknown" {
				t.Fatalf("hard unknown was accepted: %+v %v", r, e)
			}
			f.mu.Lock()
			defer f.mu.Unlock()
			max := 1
			if mode == "extra_terminal_item" {
				max = 2
			}
			if mode == "multipart" {
				max = 3
			}
			if f.creates > max || f.cancels > 1 || f.violation != "" {
				t.Fatalf("unknown continued: %+v", f)
			}
		})
	}
}

func TestV2BudgetThreeSlotsAndPhaseAggregateFour(t *testing.T) {
	p := ledgerV2(t)
	if reserve(p, "primary", attemptPrefix+"/completed-generate") == nil {
		t.Fatal("new initial completed generation must not be admitted")
	}
	for _, slot := range []string{"cancel-generate", "cancel-followup", "completed-followup"} {
		if e := reserve(p, "primary", attemptPrefix+"/"+slot); e != nil {
			t.Fatal(e)
		}
	}
	if reserve(p, "primary", attemptPrefix+"/cancel-generate") == nil {
		t.Fatal("unknown intent retried")
	}
	// A second consumed v1 primary leaves only two slots for v2.
	p = ledgerV2(t)
	f, e := os.OpenFile(p, os.O_APPEND|os.O_WRONLY, 0600)
	if e != nil {
		t.Fatal(e)
	}
	json.NewEncoder(f).Encode(map[string]any{"event": "intent", "stage": "primary", "attempt": "generated-audio-repair-v1/completed-followup"})
	f.Close()
	for _, slot := range []string{"cancel-generate", "cancel-followup"} {
		if e := reserve(p, "primary", attemptPrefix+"/"+slot); e != nil {
			t.Fatal(e)
		}
	}
	if reserve(p, "primary", attemptPrefix+"/completed-followup") == nil {
		t.Fatal("phase aggregate exceeded four primary intents")
	}
}

func TestV2AuditIDsRemainUnique(t *testing.T) {
	u, _ := serverV2(t, "normal")
	out := filepath.Join(t.TempDir(), "protocol-v2")
	if _, e := runProbe(context.Background(), probeOptions{URL: u, Ledger: ledgerV2(t), Output: out, StepTimeout: time.Second}); e != nil {
		t.Fatal(e)
	}
	f, e := os.Open(filepath.Join(out, "events.jsonl"))
	if e != nil {
		t.Fatal(e)
	}
	defer f.Close()
	s := bufio.NewScanner(f)
	seen := map[string]bool{}
	for s.Scan() {
		var row map[string]any
		json.Unmarshal(s.Bytes(), &row)
		if row["direction"] != "receive" {
			continue
		}
		id := identity(row["event_id"])
		if id == "" || seen[id] {
			t.Fatal("missing/duplicate audit ID")
		}
		seen[id] = true
	}
	if s.Err() != nil {
		t.Fatal(s.Err())
	}
}
