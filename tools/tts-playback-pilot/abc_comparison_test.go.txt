type abcRun func(context.Context, string, string, []byte, string) (voicecompare.Result, error)
type abcSource struct {
	Text               string

}

type abcBoundary struct {

	Cut, Ended, Final bool

type abcAlignment struct {
	SourcePlayed int64  `json:"source_played_samples"`

func abcScope(allow, route string) error {
	if allow != abcAllow || route != "A" && route != "B" && route != "C" {
		return errors.New("abc_explicit_route_required")
	}
	return nil
}

// Only private intent/result files receive provider errors. Keep cancellation
// distinct and bound diagnostics that may contain a provider's rejected text.
func abcPrivateFailure(err error) any {
	if err == nil {
		return nil
	}
	reason := "failed_or_unknown"
	if errors.Is(err, context.Canceled) {
		reason = "cancelled"
	} else if errors.Is(err, context.DeadlineExceeded) {
		reason = "deadline_exceeded"
	}
	text := []rune(err.Error())
	return map[string]any{"reason": reason, "message": string(text[:min(2048, len(text))]), "truncated": len(text) > 2048}
}

// Consume the existing append-only journal. All pending/failed/unknown intents
// count forever; result rows never refund them. The lock and fsync precede calls.
func abcReserve(ctx context.Context, path, route, sid string, id int64) error {
	if abcScope(abcAllow, route) != nil || len(sid) != 32 || id <= 0 {
		return errors.New("abc_invalid_attempt")
	}
	if _, e := hex.DecodeString(sid); e != nil {
		return errors.New("abc_invalid_session")
	}
	if e := ctx.Err(); e != nil {
		return e
	}
	f, e := os.OpenFile(path, os.O_RDWR|os.O_APPEND|syscall.O_NOFOLLOW, 0600)
	if e != nil {
		return e
	}
	defer f.Close()
	if e = syscall.Flock(int(f.Fd()), syscall.LOCK_EX|syscall.LOCK_NB); e != nil {
		return e
	}
	defer syscall.Flock(int(f.Fd()), syscall.LOCK_UN)
	st, e := f.Stat()
	if e != nil {
		return e
	}
	if !st.Mode().IsRegular() || st.Mode().Perm() != 0600 || st.Size() > 8<<20 {
		return errors.New("abc_invalid_ledger")
	}
	if st.Size() > 0 {
		var tail [1]byte
		if _, e = f.ReadAt(tail[:], st.Size()-1); e != nil || tail[0] != '\n' {
			return errors.New("abc_ledger_tail_unknown")
		}
	}
	attempt := fmt.Sprintf("%s%s/%s/turn-%d", abcPhase, route, sid, id)
	primary, phase, own := 0, 0, 0
	seen := map[string]bool{}
	scan := bufio.NewScanner(f)
	scan.Buffer(make([]byte, 4096), 1<<20)
	for scan.Scan() {
		var row map[string]any
		if json.Unmarshal(scan.Bytes(), &row) != nil {
			return errors.New("abc_ledger_corrupt")
		}
		event, ok := row["event"].(string)
		if !ok || event != "intent" && event != "result" {
			return errors.New("abc_ledger_event_unknown")
		}
		if event == "result" {
			prior, ok := row["attempt"].(string)
			if !ok || !seen[prior] || row["stage"] != "primary" && row["stage"] != "protocol" {
				return errors.New("abc_result_history_unknown")
			}
			continue
		}
		prior, ok := row["attempt"].(string)
		if !ok || prior == "" || seen[prior] || prior == attempt {
			return errors.New("abc_duplicate_or_unknown")
		}
		seen[prior] = true
		switch row["stage"] {
		case "protocol":
		case "primary":
			primary++
		default:
			return errors.New("abc_ledger_stage_unknown")
		}
		if strings.HasPrefix(prior, abcPhase) {
			parts := strings.Split(strings.TrimPrefix(prior, abcPhase), "/")
			if len(parts) != 3 || abcScope(abcAllow, parts[0]) != nil || len(parts[1]) != 32 || !strings.HasPrefix(parts[2], "turn-") || row["stage"] != "primary" {
				return errors.New("abc_phase_history_unknown")
			}
			if _, e = hex.DecodeString(parts[1]); e != nil {
				return errors.New("abc_phase_history_unknown")
			}
			var turn int64
			if _, e = fmt.Sscanf(parts[2], "turn-%d", &turn); e != nil || turn <= 0 || fmt.Sprintf("turn-%d", turn) != parts[2] {
				return errors.New("abc_phase_history_unknown")
			}
			phase++
			if parts[0] == route {
				own++
			}
		}
	}
	if e = scan.Err(); e != nil {
		return e
	}
	if primary >= 50 || phase >= 12 || own >= 4 {
		return errors.New("abc_primary_limit")
	}
	if e = ctx.Err(); e != nil {
		return e
	}
	raw, e := json.Marshal(map[string]any{"event": "intent", "stage": "primary", "attempt": attempt, "backend": "abc-" + route, "state": "outcome_unknown", "at": time.Now().UTC().Format(time.RFC3339Nano)})
	if e != nil {
		return e
	}
	raw = append(raw, '\n')
	n, e := f.Write(raw)
	if e != nil {
		return e
	}
	if n != len(raw) {
		return io.ErrShortWrite
	}
	return f.Sync()
}
func abcAdmit(s *Session) error {
	w, ok := s.wire.(*xiaozhiWire)
	if s.ownerErr != nil || s.rec.owner.DeviceID != abcOwner || s.rec.owner.ClientType != "xiaozhi" || !ok || w.pcm {
		return errors.New("abc_target_opus_required")
	}
	return nil
}
func abcModel(route string) string {
	if route == "B" {
		return "qwen-plus"
	}
	return "qwen3.8-omni-flash-realtime"
}
func abcHealth(s *Server, route string) map[string]any {
	s.mu.Lock()
	active, ready := len(s.leases), !s.stopping && s.initErr == nil
	s.mu.Unlock()
	tts, voice, mode, inputASR := "cosyvoice-v3-flash", "longanyang", "buffered_complete_tts", "none"
	if route == "B" {
		inputASR = "fun-asr-realtime"
	}
	if route == "C" {
		tts, voice, mode = "native", s.cfg.Voice, "native_streaming"
	}
	return map[string]any{"ready": ready, "build": abcBuild, "route": route, "primary_model": abcModel(route), "model": abcModel(route), "input_asr_model": inputASR, "common_input_listener": "qwen3-asr-flash-realtime", "output_asr_model": "fun-asr-realtime", "tts_model": tts, "voice": voice, "mode": mode, "target_device": abcOwner, "target_mac": abcMAC, "transport": "opus", "max_sessions": s.cfg.MaxSessions, "active_sessions": active, "native_conversation": false, "memory_enabled": s.cfg.MemoryEnabled, "diagnostics": s.cfg.DiagnosticsDevice != "", "parallel_planning": s.cfg.ParallelPlanning, "music_online": s.cfg.Music != nil || s.cfg.MusicForDevice != nil, "expressions": s.cfg.ExpressionsEnabled, "tools": len(s.cfg.Tools) > 0, "speaker": s.cfg.Speaker != nil, "search": s.cfg.Search != nil && *s.cfg.Search, "output_asr": true}
}
func abcHandler(s *Server, route string, next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		host, _, e := net.SplitHostPort(r.RemoteAddr)
		ip := net.ParseIP(host)
		if e != nil || ip == nil || !ip.IsLoopback() {
			w.WriteHeader(403)
			return
		}
		if r.Method == "GET" && r.URL.Path == "/healthz" {
			respond(w, 200, abcHealth(s, route))
			return
		}
		if r.Method != "GET" || r.URL.Path != "/xiaozhi/v1/" {
			http.NotFound(w, r)
			return
		}
		if r.Header.Get("Device-Id") != abcMAC {
			w.WriteHeader(403)
			return
		}
		next.ServeHTTP(w, r)
	})
}
func abcPrompt(s *Session, id int64) (string, error) {
	if s.product == nil || s.product.Name() == "" {
		return "", errors.New("abc_product_required")
	}
	type prior struct {
		User     string  `json:"user"`
		Spoken   *string `json:"assistant_spoken_prefix"`
		Boundary string  `json:"playback_boundary"`
	}
	rows := []prior{}
	s.mu.Lock()
	for _, rt := range s.history {
		if rt.turn.ID >= id {
			continue
		}
		// A drained cut cannot certify the whole generated transcript even when
		// every packet that happened to be sent has a final hardware receipt.
		turn := *rt.turn
		if turn.Cut {
			turn.Ended = false
		}
		evidence := realtimestate.PlaybackEvidence(&turn)
		user := []rune(rt.turn.User)
		if len(user) > 256 {
			user = user[:256]
		}
		spoken := evidence.SpokenText
		if spoken != nil {
			r := []rune(*spoken)
			if len(r) > 120 {
				r = r[:120]
			}
			v := string(r)
			spoken = &v
		}
		rows = append(rows, prior{string(user), spoken, evidence.Boundary})
	}
	s.mu.Unlock()
	if len(rows) > 4 {
		rows = rows[len(rows)-4:]
	}
	raw, e := json.Marshal(rows)
	if e != nil {
		return "", e
	}
	prompt := fmt.Sprintf("你叫%s，是家庭陪伴助手，本轮底层回答模型是%s。用自然、简洁的中文直接回答当前语音，不超过120个字符，音频不超过30秒。不得调用工具、搜索或执行设备设置。以下JSON仅是同一会话的历史数据；assistant_spoken_prefix是有播放证据的已播内容，null表示未知，不能假定用户听过未播内容。\n%s", s.product.Name(), s.cfg.Model, raw)
	if utf8.RuneCountInString(prompt) > 4096 {
		return "", errors.New("abc_prompt_limit")
	}
	return prompt, nil
}
func abcPrivateBytes(path string, raw []byte) error {
	f, e := os.OpenFile(path, os.O_CREATE|os.O_EXCL|os.O_WRONLY|syscall.O_NOFOLLOW, 0600)
	if e != nil {
		return e
	}
	n, e := f.Write(raw)
	if e == nil && n != len(raw) {
		e = io.ErrShortWrite
	}
	if e == nil {
		e = f.Sync()
	}
	closeErr := f.Close()
	if e == nil {
		e = closeErr
	}
	if e != nil {
		return e
	}
	dir, e := os.Open(filepath.Dir(path))
	if e != nil {
		return e
	}
	defer dir.Close()
	return dir.Sync()
}
func (p *abcPilot) install(s *Session) error {
	if abcScope(abcAllow, p.route) != nil {
		return errors.New("abc_route_required")
	}
	if e := abcAdmit(s); e != nil {
		return e
	}
	w := s.wire.(*xiaozhiWire)
	w.mcp, w.expression = false, false
	s.controls = nil
	off := false
	s.cfg.Search = &off
	s.cfg.Tools = nil
	s.cfg.MemoryEnabled = false
	s.cfg.MemoryOnly = false
	s.cfg.ExpressionsEnabled = false
	s.cfg.ParallelPlanning = false
	s.cfg.Music = nil
	s.cfg.MusicForDevice = nil
	s.cfg.Speaker = nil
	s.cfg.DiagnosticsDevice = ""
	s.memory = nil
	s.diag = nil
	s.listener.cfg = s.cfg.snapshot()
	s.cfg.Model = abcModel(p.route)
	s.generate = func(ctx context.Context, id int64, _ string, pcm []byte, emit func(string, []byte) error) (err error) {
		if e := ctx.Err(); e != nil {
			return e
		}
		if len(pcm) == 0 || len(pcm)%2 != 0 || len(pcm) > 16000*2*10 {
			return errors.New("abc_input_limit")
		}
		prompt, e := abcPrompt(s, id)
		if e != nil {
			return e
		}
		if e = abcReserve(ctx, p.ledger, p.route, s.rec.owner.SessionID, id); e != nil {
			return e
		}
		audit, e := newIntent(s.directory, "abc-primary", false)
		if e != nil {
			return e
		}
		if e = audit.update(map[string]any{"route": p.route, "turn": id, "model": abcModel(p.route), "mode": abcHealthMode(p.route), "state": "outcome_unknown", "input_samples": len(pcm) / 2, "input_sample_rate": 16000, "usage_state": "unknown"}); e != nil {
			return e
		}
		defer func() {
			state := "completed"
			if err != nil {
				state = "failed_cancelled_or_unknown"
			}
			err = errors.Join(err, audit.update(map[string]any{"state": state, "error": abcPrivateFailure(err)}))
		}()
		token := fmt.Sprintf("%s-turn-%d-%s", p.route, id, randomID())
		if e = abcPrivateBytes(filepath.Join(s.directory, "abc-input-"+token+".pcm"), pcm); e != nil {
			return e
		}
		if e = ctx.Err(); e != nil {
			return e
		}
		if e = emit("ready", nil); e != nil {
			return e
		}
		if p.route == "C" {
			received, textChars, tooLong := 0, 0, false
			err = s.generateNative(ctx, id, prompt, pcm, func(kind string, value []byte) error {
				if e := ctx.Err(); e != nil {
					return e
				}
				if kind == "pcm" || kind == "audio" {
					received += len(value) / 2
					if received > 24000*30 {
						tooLong = true
						return errors.New("abc_native_output_limit")
					}
				}
				if kind == "text" {
					textChars = utf8.RuneCount(value)
					if textChars > 120 {
						tooLong = true
						return errors.New("abc_native_text_limit")
					}
				}
				return emit(kind, value)
			})
			err = errors.Join(err, audit.update(map[string]any{"received_samples": received, "sample_rate": 24000, "text_characters": textChars, "overlong": tooLong, "usage_source": "live_native_turn_audit", "usage_state": "see_native_audit"}))
			return err
		}
		client := p.client
		events, e := os.OpenFile(filepath.Join(s.directory, "abc-events-"+token+".jsonl"), os.O_CREATE|os.O_EXCL|os.O_WRONLY|syscall.O_NOFOLLOW, 0600)
		if e != nil {
			return e
		}
		defer events.Close()
		eventBytes := 0
		client.RecordEvent = func(raw json.RawMessage) error {
			var compact bytes.Buffer
			if e := json.Compact(&compact, raw); e != nil {
				return e
			}
			eventBytes += compact.Len() + 1
			if eventBytes > 8<<20 {
				return errors.New("abc_provider_event_limit")
			}
			compact.WriteByte('\n')
			n, e := events.Write(compact.Bytes())
			if e != nil {
				return e
			}
			if n != compact.Len() {
				return io.ErrShortWrite
			}
			return events.Sync()
		}
		callIndex := 0
		var current *intent
		client.BeforeCall = func(stage string, reserve float64) error {
			if e := ctx.Err(); e != nil {
				return e
			}
			callIndex++
			var e error
			current, e = newIntent(s.directory, "abc-api", false)
			if e != nil {
				return e
			}
			models := map[string]string{"native": "qwen3.8-omni-flash-realtime", "asr": "fun-asr-realtime", "llm": "qwen-plus", "tts": "cosyvoice-v3-flash"}
			return current.update(map[string]any{"route": p.route, "turn": id, "api_stage": stage, "model": models[stage], "call_index": callIndex, "reserve_yuan_estimate": reserve, "state": "submission_outcome_unknown", "usage_state": "unknown"})
		}
		client.AfterCall = func(stage string, usage map[string]any, callErr error) error {
			if current == nil {
				return errors.New("abc_api_intent_missing")
			}
			state, usageState := "completed", "known"
			if callErr != nil {
				state = "failed_cancelled_or_unknown"
				usageState = "reported_partial_or_unknown"
			}
			if usage == nil {
				usageState = "unknown"
			}
			return current.update(map[string]any{"api_stage": stage, "state": state, "usage_state": usageState, "usage": usage, "error": abcPrivateFailure(callErr)})
		}
		// The client already validates provider events and final sentence-end words.
		// Raw events and partial transcripts are never copied to stdout or a prompt.
		run := p.run
		if run == nil {
			run = client.Run
		}
		backend := "native"
		if p.route == "B" {
			backend = "cascade"
		}
		result, callErr := run(ctx, backend, prompt, pcm, "")
		callErr = errors.Join(callErr, events.Close())
		if callErr == nil {
			callErr = abcValidate(result)
		}
		if ctx.Err() != nil {
			callErr = ctx.Err()
		}
		state := "completed"
		if callErr != nil {
			state = "failed_cancelled_or_unknown"
		}
		if len(result.PCM) > 0 {
			if e = abcPrivateBytes(filepath.Join(s.directory, "abc-output-"+token+".pcm"), result.PCM); e != nil {
				return errors.Join(callErr, e)
			}
		}
		if e = privateJSON(filepath.Join(s.directory, "abc-result-"+token+".json"), map[string]any{"route": p.route, "turn": id, "state": state, "mode": "buffered_complete_tts", "primary_model": abcModel(p.route), "tts_model": "cosyvoice-v3-flash", "voice": "longanyang", "sample_rate": 24000, "samples": len(result.PCM) / 2, "text": result.Text, "words": result.Words, "calls": result.Calls, "usage_unknown": callErr != nil, "error": abcPrivateFailure(callErr)}); e != nil {
			return errors.Join(callErr, e)
		}
		if callErr != nil {
			return callErr
		}
		if e = ctx.Err(); e != nil {
			return e
		}
		codec, e := realtimeaudio.New(24000, 1440)
		if e != nil {
			return e
		}
		delay, e := codec.LookaheadSamples()
		codec.Close()
		if e != nil {
			return e
		}
		p.mu.Lock()
		if p.sources == nil {
			p.sources = map[int64]abcSource{}
		}
		p.sources[id] = abcSource{Text: result.Text, Words: append([]voicecompare.Word(nil), result.Words...), Samples: int64(len(result.PCM) / 2), Lookahead: int64(delay)}
		p.mu.Unlock()
		if e = emit("text", []byte(result.Text)); e != nil {
			return e
		}
		if e = ctx.Err(); e != nil {
			return e
		}
		return emit("audio", result.PCM)
	}
	return nil
}
func abcHealthMode(route string) string {
	if route == "C" {
		return "native_streaming"
	}
	return "buffered_complete_tts"
}
func abcValidate(result voicecompare.Result) error {
	if strings.TrimSpace(result.Text) == "" || !utf8.ValidString(result.Text) || utf8.RuneCountInString(result.Text) > 120 || len(result.PCM) == 0 || len(result.PCM)%2 != 0 || len(result.PCM) > 24000*2*30 || len(result.Words) == 0 {
		return errors.New("abc_incomplete_result")
	}
	end := int64(0)
	var covered strings.Builder
	for _, word := range result.Words {
		if word.Text == "" || word.BeginMS < end || word.EndMS <= word.BeginMS || word.EndMS > int64(len(result.PCM)/2)/24 {
			return errors.New("abc_invalid_words")
		}
		end = word.EndMS
		covered.WriteString(word.Text)
	}
	if covered.String() != result.Text {
		return errors.New("abc_word_coverage_missing")
	}
	return nil
}

// Same encoder-lookahead and 120ms rules as fixedTTSMap. This is a read-only
// audit of the frozen receipt, never a replacement for common OutputASR context.
func abcMap(source abcSource, b abcBoundary) abcAlignment {
	result := abcAlignment{Unknown: true}
	if b.Basis != "i2s_dma" || source.Lookahead < 0 || b.Completed < 0 || b.Completed > b.Sent {
		return result
	}
	result.Hardware = true
	result.SourcePlayed = max(int64(0), min(source.Samples, b.Completed-source.Lookahead))
	result.Prefix = voicecompare.ConfirmedText(source.Words, result.SourcePlayed, 120)
	result.Full = b.Ended && !b.Cut && b.Final && b.Sent > 0 && b.Completed == b.Sent && result.SourcePlayed == source.Samples
	result.Unknown = !result.Full
	if result.Full {
		result.Prefix = source.Text
	} else if result.Prefix == source.Text && len(source.Words) > 0 {
		result.Prefix = strings.TrimSuffix(result.Prefix, source.Words[len(source.Words)-1].Text)
	}
	return result
}
func (p *abcPilot) finish(s *Session) error {
	type frozen struct {
		ID       int64
		Source   abcSource
		Boundary abcBoundary
	}
	rows := []frozen{}
	p.mu.Lock()
	s.mu.Lock()
	for _, rt := range s.history {
		if source, ok := p.sources[rt.turn.ID]; ok {
			v := rt.turn
			rows = append(rows, frozen{v.ID, source, abcBoundary{v.Completed, v.Sent, v.Cut, v.Ended, v.PlaybackFinal, v.PlaybackBasis}})
		}
	}
	p.sources = nil // The terminal snapshot cannot be upgraded by a later call.
	s.mu.Unlock()
	p.mu.Unlock()
	for _, row := range rows {
		if e := privateJSON(filepath.Join(s.directory, fmt.Sprintf("abc-alignment-%s-turn-%d.json", p.route, row.ID)), map[string]any{"route": p.route, "turn": row.ID, "source": row.Source, "boundary": row.Boundary, "alignment": abcMap(row.Source, row.Boundary), "margin_ms": 120, "audit_only": true}); e != nil {
			return e
		}
	}
	return nil
}

func abcLedger(t *testing.T, historical int) string {

				c, e := (&websocket.Upgrader{}).Upgrade(w, r, nil)
				if e != nil {
					return
				}
				defer c.Close()
				c.SetReadDeadline(time.Now().Add(3 * time.Second))
				var request map[string]any
				if c.ReadJSON(&request) != nil {
					return
				}
				header, payload := object(request["header"]), object(request["payload"])
				id := header["task_id"]
				stage, _ := payload["task"].(string)
				checkIntent(stage)
				event := func(kind string, output any) {
					usage := map[string]any{"characters": 3}
					if stage == "asr" {
						usage = map[string]any{"duration": .02}
					}
					_ = c.WriteJSON(map[string]any{"header": map[string]any{"task_id": id, "event": kind}, "payload": map[string]any{"output": output, "usage": usage}})
				}
				event("task-started", nil)
				if stage == "asr" {
					if payload["model"] != "fun-asr-realtime" {
						t.Error("wrong ASR model")
					}
					for {
						typ, raw, e := c.ReadMessage()
						if e != nil {
							return
						}
						if typ == websocket.TextMessage {
							var v map[string]any
							_ = json.Unmarshal(raw, &v)
							if object(v["header"])["action"] == "finish-task" {
								break
							}
						}
					}
					event("result-generated", map[string]any{"sentence": map[string]any{"text": "你好", "sentence_end": true}})
					event("task-finished", nil)
					return
				}
				if stage != "tts" || payload["model"] != "cosyvoice-v3-flash" || object(payload["parameters"])["voice"] != "longanyang" {
					t.Error("wrong TTS route/model/voice")
				}
				if c.ReadJSON(&request) != nil {
					return
				}
				text, _ := object(object(request["payload"])["input"])["text"].(string)
				if c.ReadJSON(&request) != nil {
					return
				}
				out := func(kind string, words any) map[string]any {
					return map[string]any{"type": kind, "original_text": text, "sentence": map[string]any{"index": 0, "words": words}}
				}
				event("result-generated", out("sentence-begin", nil))
				event("result-generated", out("sentence-synthesis", nil))
				_ = c.WriteMessage(websocket.BinaryMessage, make([]byte, 24000))
				words := []any{map[string]any{"text": text, "begin_index": 0, "end_index": len([]rune(text)), "begin_time": 0, "end_time": 400}}
				event("result-generated", out("sentence-end", words))
				event("task-finished", nil)
			}))
			defer inference.Close()
			textAPI := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
				checkIntent("llm")
				var request map[string]any
				if json.NewDecoder(r.Body).Decode(&request) != nil {
					t.Error("bad JSON")
				}
				if request["model"] != "qwen-plus" || request["enable_thinking"] != false || request["enable_search"] != false {
					t.Error("wrong Plus mode")
				}
				if label == "B-http-rejected" {
					w.WriteHeader(http.StatusServiceUnavailable)
					_, _ = fmt.Fprint(w, `{"error":{"code":"fixture_rejected"}}`)
					return
				}
				_ = json.NewEncoder(w).Encode(map[string]any{"choices": []any{map[string]any{"finish_reason": "stop", "message": map[string]any{"role": "assistant", "content": "好呀。"}}}, "usage": map[string]any{"prompt_tokens": 4, "completion_tokens": 3, "total_tokens": 7}})
			}))
			defer textAPI.Close()
			nativeURL := "ws" + strings.TrimPrefix(native.URL, "http")
			s.cfg.URL = nativeURL
			s.listener.cfg.URL = nativeURL
			p := &abcPilot{route: route, ledger: abcLedger(t, 38), client: voicecompare.Client{Key: "fixture", RealtimeURL: nativeURL, InferenceURL: "ws" + strings.TrimPrefix(inference.URL, "http"), TextURL: textAPI.URL, TTSModel: "cosyvoice-v3-flash", Voice: "longanyang"}}
			if e := p.install(s); e != nil {
				t.Fatal(e)
			}
			var audio int
			e := s.generate(ctx, 1, "unplayed diagnostic", make([]byte, 640), func(k string, v []byte) error {
				if k == "audio" || k == "pcm" {
					audio += len(v)
				}
				return nil
			})
			if (e != nil) != fail || fail && audio != 0 || !fail && audio == 0 {
				t.Fatal("incorrect terminal/audio", e, audio, fail)
			}
			close(stages)
			var got []string
			for stage := range stages {
				got = append(got, stage)
			}
			want := "native,tts"
			if route == "B" {
				want = "asr,llm,tts"
			}
			if route == "C" || fail {
				want = "native"
			}
			if label == "B-http-rejected" {
				want = "asr,llm"
			}
			if strings.Join(got, ",") != want {
				t.Fatal("crossed backend", got, want)
			}
			if route != "C" {
				paths, _ := filepath.Glob(filepath.Join(s.directory, "abc-api-*.json"))
				if len(paths) != len(got) {
					t.Fatal("missing API usage records", paths)
				}
				for _, path := range paths {
					raw, _ := os.ReadFile(path)
					var row map[string]any
					_ = json.Unmarshal(raw, &row)
					callFailed := fail && (label != "B-http-rejected" || row["api_stage"] == "llm")
					if !callFailed && (row["state"] != "completed" || row["usage_state"] != "known" || row["usage"] == nil) || callFailed && (row["state"] != "failed_cancelled_or_unknown" || row["usage_state"] != "unknown") {
						t.Fatal("usage was lost or unknown became success", row)
					}
					if callFailed {
						failure := object(row["error"])
						detail, _ := failure["message"].(string)
						if failure["reason"] == nil || detail == "" || utf8.RuneCountInString(detail) > 2048 || label == "B-http-rejected" && !strings.Contains(detail, "text provider rejected request") {
							t.Fatal("API intent omitted private failure cause")
						}
					}
				}
				if fail {
					paths, _ := filepath.Glob(filepath.Join(s.directory, "abc-result-*.json"))
					if len(paths) != 1 {
						t.Fatal("missing failed result")
					}
					raw, _ := os.ReadFile(paths[0])
					var result map[string]any
					_ = json.Unmarshal(raw, &result)
					if object(result["error"])["message"] == nil {
						t.Fatal("result omitted provider failure cause")
					}
				}
				events, _ := filepath.Glob(filepath.Join(s.directory, "abc-events-*.jsonl"))
				if len(events) != 1 {
					t.Fatal(events)
				}
				raw, _ := os.ReadFile(events[0])
				if len(raw) == 0 || raw[len(raw)-1] != '\n' {
					t.Fatal("missing raw provider evidence")
				}
			} else if fail {
				paths, _ := filepath.Glob(filepath.Join(s.directory, "abc-primary-*.json"))
				if len(paths) != 1 {
					t.Fatal("missing overlong native audit", paths)
				}
				raw, _ := os.ReadFile(paths[0])
				var row map[string]any
				if json.Unmarshal(raw, &row) != nil || row["overlong"] != true || row["state"] != "failed_cancelled_or_unknown" || row["received_samples"] != float64(24000*30+1) {
					t.Fatal("overlong native was presented as completed", row)
				}
				if detail, _ := object(row["error"])["message"].(string); !strings.Contains(detail, "abc_native_output_limit") {
					t.Fatal("native primary omitted private limit cause")
				}
			}
		})
	}
}

func TestABCInvalidInputAndPrivateLedgerNeverSubmit(t *testing.T) {
	s := abcSession(t)
	path := abcLedger(t, 38)
	calls := 0
	p := &abcPilot{route: "C", ledger: path, run: func(context.Context, string, string, []byte, string) (voicecompare.Result, error) {
		calls++
		return abcGoodResult(), nil
	}}
	if e := p.install(s); e != nil {
		t.Fatal(e)
	}
	before, _ := os.ReadFile(path)
	for _, pcm := range [][]byte{nil, make([]byte, 1), make([]byte, 16000*2*10+2)} {
		if e := s.generate(context.Background(), 1, "", pcm, func(string, []byte) error { t.Error("invalid input emitted event"); return nil }); e == nil {
			t.Fatal("invalid input accepted")
		}
	}
	after, _ := os.ReadFile(path)
	if calls != 0 || string(before) != string(after) {
		t.Fatal("invalid input reserved or submitted")
	}
	if e := os.Chmod(path, 0644); e != nil {
		t.Fatal(e)
	}
	if abcReserve(context.Background(), path, "A", abcTestSID, 1) == nil {
		t.Fatal("public ledger accepted")
	}
	if e := os.Chmod(path, 0600); e != nil {
		t.Fatal(e)
	}
	link := filepath.Join(t.TempDir(), "linked-ledger")
	if e := os.Symlink(path, link); e != nil {
		t.Fatal(e)
	}
	if abcReserve(context.Background(), link, "A", abcTestSID, 1) == nil {
		t.Fatal("symlink ledger accepted")
	}
}
